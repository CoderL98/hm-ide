@Entry
@Component
struct Index {
@State message: string = 'Hello HarmonyOS';
@State count: number = 0;
 
build() {
Column() {
Text(this.message)
.fontSize(28)
.fontWeight(FontWeight.Medium)
.margin({ bottom: 16 })
 
Text(`点击次数：${this.count}`)
.fontSize(16)
.fontColor('#A1A1AA')
.margin({ bottom: 24 })
 
Button('增加')
.onClick(() => {
this.count += 1;
})
.width(120)
}
.width('100%')
.height('100%')
.justifyContent(FlexAlign.Center)
}
}
